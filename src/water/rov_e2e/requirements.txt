# rov_e2e 依赖清单
#
# 目标环境：Ubuntu 20.04（教学虚拟机）+ ROS 2 Humble + Python 3.8
# 安装：
#   pip3 install -r src/water/rov_e2e/requirements.txt
#   # 国内加速：-i https://pypi.tuna.tsinghua.edu.cn/simple
#
# ---------------------------------------------------------------------------
# 深度/强化学习框架（CPU 版本即可，虚拟机无独立显卡）
#
# ⚠ Python 3.8 版本上限很关键：
#   PyTorch 自 2.5.0 起不再提供 Python 3.8 的 wheel，最后一个支持 3.8 的
#   版本是 torch 2.4.1 / torchvision 0.19.1。若直接 `pip3 install torch`
#   在 Python 3.8 上会解析失败或装到过旧版本。
#   若使用 Python 3.10（Ubuntu 22.04），可放宽为 torch>=2.0。
# ---------------------------------------------------------------------------
torch==2.4.1
torchvision==0.19.1

# 图像读写（采集脚本存 PNG、训练脚本读 PNG）
pillow>=8.0.0
numpy>=1.20.0

# 训练损失曲线与离线测试散点图（可选，缺失时脚本自动跳过绘图）
matplotlib>=3.3.0

# MuJoCo 物理引擎。注意：mujoco>=3.2 起要求 Python>=3.9，
# 因此在 Python 3.8 的教学虚拟机上请使用 3.0/3.1 系列。
mujoco>=3.0.0,<3.2
